from gpiozero import LED
from time import sleep

led = LED(17)

# Pattern 1: Slow blinking
print("Pattern 1: Slow blinking")

for i in range(5):
    led.on()
    sleep(1)
    led.off()
    sleep(1)

# Pattern 2: Fast blinking
print("Pattern 2: Fast blinking")

for i in range(10):
    led.on()
    sleep(0.2)
    led.off()
    sleep(0.2)

# Pattern 3: Long ON, short OFF
print("Pattern 3: Long ON, short OFF")

for i in range(5):
    led.on()
    sleep(2)
    led.off()
    sleep(0.5)

led.off()
print("All patterns completed!")
